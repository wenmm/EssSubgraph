import numpy as np
from sklearn.metrics import precision_recall_curve, f1_score


# 计算 Precision, Recall, Threshold
precision, recall, thresholds = precision_recall_curve(true_labels, predicted_probs)

# 由于 precision_recall_curve 的返回值不包括 threshold=0 的情况，所以我们补齐对应长度
f1 = 2 * (precision * recall) / (precision + recall + 1e-8)  # 防止除零

# 找到最大 F1 的阈值索引
best_f1_idx = np.argmax(f1)
best_f1_threshold = thresholds[best_f1_idx]
best_f1_value = f1[best_f1_idx]

# 找到最大 Precision 的阈值索引（排除 recall=0 的点）
best_prec_idx = np.argmax(precision[:-1])  # 最后一个点的 precision 没有对应 threshold
best_prec_threshold = thresholds[best_prec_idx]

# 找到最大 Recall 的阈值索引（第一个点通常 recall=1）
best_recall_idx = np.argmax(recall[:-1])
best_recall_threshold = thresholds[best_recall_idx]

print(f"Best F1: {best_f1_value:.4f} at threshold = {best_f1_threshold:.2f}")
print(f"Best Precision at threshold = {best_prec_threshold:.2f}")
print(f"Best Recall at threshold = {best_recall_threshold:.2f}")

import numpy as np
from sklearn.metrics import roc_curve


# 计算 ROC 曲线数据
fpr, tpr, thresholds = roc_curve(true_labels, predicted_probs)

# 找到 FPR 最小的索引集合
min_fpr = np.min(fpr)
min_fpr_indices = np.where(fpr == min_fpr)[0]

# 在这些 index 中选择 TPR 最大的那个
best_idx = min_fpr_indices[np.argmax(tpr[min_fpr_indices])]

# 输出最佳阈值
best_threshold = thresholds[best_idx]
print(f"Best cutoff (min FPR, max TPR) = {best_threshold:.4f}")
print(f"FPR = {fpr[best_idx]:.4f}, TPR = {tpr[best_idx]:.4f}")

import numpy as np
from sklearn.metrics import roc_curve
import matplotlib.pyplot as plt


# ROC 曲线
fpr, tpr, thresholds = roc_curve(true_labels, predicted_probs)

# 找到最大 TPR - FPR 的位置（转弯点）
youden_j = tpr - fpr
best_idx = np.argmax(youden_j)
best_threshold = thresholds[best_idx]


print(f"Best turning point threshold = {best_threshold:.4f}")
print(f"TPR = {tpr[best_idx]:.4f}, FPR = {fpr[best_idx]:.4f}")

# 可视化
plt.plot(fpr, tpr, label='ROC curve')
plt.scatter(fpr[best_idx], tpr[best_idx], color='red', label='Turning point')
plt.plot([0, 1], [0, 1], linestyle='--', color='gray')
plt.xlabel('FPR')
plt.ylabel('TPR')
plt.title('ROC Curve with Turning Point')
plt.legend()
plt.grid(True)

plt.xlim(0, 1)
plt.ylim(0, 1)

plt.savefig('roc_curve.pdf', dpi=600)  # 保存为高清PDF
